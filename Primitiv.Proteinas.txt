def test_tpsa_final(datos, base_A, ref_L0):
    resultados = []
    residuos = []
    
    for nombre, valor in datos.items():
        if valor <= 0: continue
        razon_log = math.log(valor / ref_L0)
        q_entero = round(razon_log / base_A)
        
        residuo = abs(razon_log - (q_entero * base_A))
        val_teorico = ref_L0 * math.exp(q_entero * base_A)
        error_rel = abs(valor - val_teorico) / valor * 100
        
        resultados.append({
            "Molecula": nombre, "TPSA": valor, "Nodo (q)": q_entero,
            "Residuo": residuo, "Error %": error_rel
        })
        residuos.append(residuo) 
    
    return pd.DataFrame(resultados), np.mean(residuos)

# Ejecución
df_res, res_medio = test_tpsa_final(mix_data, A, L0)

# Monte Carlo (10,000 iteraciones)
iteraciones = 10000
exitos = 0
v_min, v_max = min(mix_data.values()), max(mix_data.values())

for _ in range(iteraciones):
    random_v = np.random.uniform(v_min, v_max, len(mix_data))
    r_mean = np.mean([abs(math.log(v/L0) - round(math.log(v/L0)/A)*A) for v in random_v])
    if r_mean <= res_medio:
        exitos += 1
